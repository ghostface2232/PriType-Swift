import Cocoa
import Testing
import PriTypeIMKHarness
@testable import PriTypeCore

@Suite("Review reproduction probes", .serialized)
@MainActor
struct ReviewReproductionProbes {
    @Test func legitimateRapidRepeatedKeys() {
        let h = IMKHarness(); defer { h.finish() }
        let f = h.makeField(); h.focus(f); h.keyInterval = 0.03
        h.type("rr"); h.click()
        #expect(f.client.text == "ㄱㄱ")
    }

    @Test func toggleArrivesBeforeAnOlderIMKKey() async {
        let h = IMKHarness(); defer { h.finish() }
        let f = h.makeField(); h.focus(f)
        let older = h.makeEvent(keyCode: 15, characters: "r")
        h.toggleFromKeyMonitor()
        // Let the actual scheduled main-queue block run before the older IMK event.
        await withCheckedContinuation { (continuation: CheckedContinuation<Void, Never>) in
            DispatchQueue.main.async { continuation.resume() }
        }
        let handled = f.controller.handle(older, client: f.client)
        if !handled { f.client.performHostAction(for: older) }
        #expect(f.client.text == "ㄱ")
    }

    @Test func directToMarkedWhileComposing() {
        let config = ConfigurationManager.shared
        let saved = config.experimentalDirectInsertion
        defer { config.experimentalDirectInsertion = saved }
        config.experimentalDirectInsertion = true
        let composer = HangulComposer()
        let client = FakeTextClient()
        let context = ClientContext(bundleId: client.bundleID, hasTextInputCapability: true,
                                    isLikelyDesktopArea: false, documentAccessSafe: true)
        let session = InputSession(client: client, context: context, composer: composer)
        func key(_ code: UInt16, _ chars: String) {
            let event = NSEvent.keyEvent(with: .keyDown, location: .zero, modifierFlags: [],
                timestamp: 1, windowNumber: 0, context: nil, characters: chars,
                charactersIgnoringModifiers: chars, isARepeat: false, keyCode: code)!
            _ = composer.handle(event, delegate: session.adapter)
        }
        key(15, "r"); key(40, "k")
        #expect(client.text == "가")
        config.experimentalDirectInsertion = false
        session.ensureAdapterMatchesPolicy()
        key(4, "h") // next vowel should commit 가 and start ㅗ
        #expect(client.text == "가ㅗ")
    }

    @Test func doubleSpaceAfterSilentCaretMove() {
        let h = IMKHarness(); defer { h.finish() }
        let f = h.makeField(); h.focus(f)
        h.type("rk ")
        #expect(f.client.text == "가 ")
        f.client.placeCaret(at: 1) // no marked text, so no IMK commit callback
        h.press(.space)
        #expect(f.client.text == "가  ")
    }

    @Test func replayDoesNotDuplicateCommittedSyllables() {
        let h = IMKHarness(); defer { h.finish() }
        let f = h.makeField(); h.focus(f)
        h.type("rkrkr")
        #expect(f.client.text == "가각")
        h.press(.backspace)
        #expect(f.client.text == "가가")
    }

    @Test func noSpaceLostWhenCaretIsUnavailable() {
        let convenience = TextConvenienceHandler(isDoubleSpacePeriodEnabled: { true })
        let client = FakeTextClient()
        let adapter = MarkedTextAdapter(client: client, bundleId: client.bundleID)
        var buffer = "가"
        _ = convenience.handleDoubleSpacePeriod(buffer: &buffer, delegate: adapter)
        buffer += " "
        client.select(NSRange(location: NSNotFound, length: 0))
        let result = convenience.handleDoubleSpacePeriod(buffer: &buffer, delegate: adapter)
        #expect(result == .normalSpace)
    }
}
